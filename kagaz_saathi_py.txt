HERE = os.path.dirname(os.path.abspath(__file__))
LOG = os.path.join(HERE, "bench.jsonl")

PATTERNS = [  # order matters: longer/specific first
    ("AADHAAR", r"\b\d{4}[ -]?\d{4}[ -]?\d{4}\b"),
    ("PAN", r"\b[A-Z]{5}\d{4}[A-Z]\b"),
    ("EMAIL", r"[\w.+-]+@[\w-]+\.[\w.-]+"),
    ("PHONE", r"(?:\+91[ -]?)?\b[6-9]\d{9}\b"),
    ("ACCOUNT", r"\b\d{9,18}\b"),
]

def redact(text):
    counts = {}
    for name, pat in PATTERNS:
        text, n = re.subn(pat, "[%s]" % name, text)
        if n:
            counts[name] = n
    return text, counts

LANGS = {"hi": "simple Hindi (Devanagari)", "en": "simple English",
         "hinglish": "Hinglish (Hindi in Roman script)"}

def system_prompt(lang):
    return ("You help ordinary people in India understand official documents "
            "(tax and bank notices, insurance, rent agreements, lab reports). "
            "Reply in %s, at a Class 8 reading level, with exactly these headings: "
            "1) Yeh kya hai / What this is  2) Important dates and amounts  "
            "3) What you must do  4) Warning signs. "
            "Tokens like [AADHAAR] or [PAN] are masked on purpose; never guess them. "
            "Only use facts present in the document. If unsure, say so. "
            "You are not a lawyer; suggest a professional for high-stakes decisions."
            % LANGS.get(lang, LANGS["en"]))

class H(BaseHTTPRequestHandler):
    def _json(self, obj, code=200):
        b = json.dumps(obj).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(b)))
        self.end_headers()
        self.wfile.write(b)

    def _body(self):
        return json.loads(self.rfile.read(int(self.headers.get("Content-Length", 0))) or b"{}")

    def log_message(self, *a):
        pass

    def do_GET(self):
        if self.path in ("/", "/index.html"):
            b = INDEX_HTML.encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(b)))
            self.end_headers()
            self.wfile.write(b)
        elif self.path == "/api/bench":
            rows = []
            if os.path.exists(LOG):
                rows = [json.loads(l) for l in open(LOG, encoding="utf-8") if l.strip()]
            self._json({"backend": BACKEND, "model": LLM_MODEL, "rows": rows[-20:]})
        elif self.path == "/api/health":
            try:
                urllib.request.urlopen(LLM_URL + "/models", timeout=3)
                self._json({"ok": True, "backend": BACKEND, "url": LLM_URL})
            except Exception as e:
                self._json({"ok": False, "backend": BACKEND, "url": LLM_URL, "error": str(e)})
        else:
            self.send_error(404)

    def do_POST(self):
        d = self._body()
        text, counts = redact(d.get("text", ""))
        if self.path == "/api/redact":
            return self._json({"text": text, "counts": counts})
        if self.path != "/api/explain":
            return self.send_error(404)
        payload = {"model": LLM_MODEL, "stream": True, "temperature": 0.3, "max_tokens": 700,
                   "messages": [{"role": "system", "content": system_prompt(d.get("lang", "en"))},
                                {"role": "user", "content": text[:6000]}]}
        req = urllib.request.Request(LLM_URL + "/chat/completions", json.dumps(payload).encode(),
                                     {"Content-Type": "application/json"})
        self.send_response(200)
        self.send_header("Content-Type", "text/event-stream")
        self.send_header("Cache-Control", "no-cache")
        self.end_headers()
        t0, ttft, n = time.perf_counter(), None, 0
        try:
            with urllib.request.urlopen(req, timeout=120) as r:
                for line in r:
                    if line.startswith(b"data:") and b"[DONE]" not in line:
                        if ttft is None:
                            ttft = time.perf_counter() - t0
                        n += 1  # one SSE chunk is roughly one token
                    self.wfile.write(line)
                    self.wfile.flush()
            dur = time.perf_counter() - t0
            gen = max(dur - (ttft or 0), 1e-6)
            stats = {"backend": BACKEND, "model": LLM_MODEL, "ttft_ms": round((ttft or 0) * 1000),
                     "tokens": n, "tps": round(n / gen, 1), "chars_in": len(text),
                     "masked": counts, "ts": int(time.time())}
            open(LOG, "a", encoding="utf-8").write(json.dumps(stats) + "\n")
            self.wfile.write(("data: " + json.dumps({"kc_stats": stats}) + "\n\n").encode())
        except Exception as e:
            err = {"kc_error": "Could not reach the local model at %s (%s). Start it first." % (LLM_URL, e)}
            self.wfile.write(("data: " + json.dumps(err) + "\n\n").encode())
